"""
### DBT_DATABRICKS_PIPELINE

Runs the **entire `dbt_databricks` project on Databricks** in dependency order.

**How the graph is built.** The project has ~5,500 models, so one Airflow task per model is not
practical (every dbt call parses the project). Instead there is one task per *domain/layer slice*
(`models/<domain>/<layer>`, plus the `dbt_artifacts` package) and the edges between slices come
straight from the dbt `ref()` graph (`include/Configuration/dbt_databricks_graph.json`, generated by
`include/Python/generate_dbt_databricks_graph.py`). Every cross-slice model dependency is enforced by
an Airflow edge; dbt orders the models *inside* a slice itself. Slices that depend on each other in
both directions (e.g. `teh_finance` silver <-> gold) are merged into one task.

**Execution.** Default: dbt runs from `dbt_venv` on the Airflow worker and sends SQL to the Databricks
SQL warehouse. Set `DBT_EXECUTION_MODE=databricks_job` to submit each slice as a Databricks Jobs
`dbt_task` instead (see `include/Python/dbt_databricks_job_operator.py`).

**Run parameters** (Trigger DAG w/ config)

| param | meaning |
|---|---|
| `dbt_command` | `run` (default), `build` (models + tests) or `test` |
| `select_nodes` | comma list of task ids or domains to run, e.g. `tea_service,teh_finance`; empty = all. Unselected upstream tasks are treated as done. |
| `skip_nodes` | comma list of task ids/domains to skip, e.g. `dbt_artifacts` |
| `full_refresh` | pass `--full-refresh` (rebuilds incremental models) |
| `threads` | dbt threads per task (mind SQL warehouse concurrency) |
| `exclude` | extra dbt `--exclude` selector |
| `dbt_vars` | extra dbt `--vars` as YAML/JSON |
| `fail_fast` | pass `--fail-fast` |
| `use_dbt_retry` | on an Airflow retry, only re-run what failed (`dbt retry`) |
| `skip_preflight` / `strict_graph_check` | see the `preflight` task |

**Deployment variables**: see `DEPLOYMENT_GUIDE.txt` (DATABRICKS_* variables, `DBT_EXECUTION_MODE`,
`DBT_MAX_PARALLEL_NODES`, `DBT_DATABRICKS_PIPELINE_SCHEDULE`).
"""
from __future__ import annotations

import logging
import os
from datetime import timedelta

import pendulum

from include.Python import dbt_databricks_common as C
from include.Python.dbt_databricks_airflow_compat import DAG, EmptyOperator, Param, TaskGroup, TriggerRule, task
from include.Python.dbt_databricks_operators import DbtDatabricksBashOperator

log = logging.getLogger("airflow.task")

EXECUTION_MODE = os.environ.get("DBT_EXECUTION_MODE", "airflow_cli").strip().lower()
MAX_PARALLEL_NODES = int(os.environ.get("DBT_MAX_PARALLEL_NODES", "8"))
NODE_TIMEOUT_HOURS = float(os.environ.get("DBT_NODE_TIMEOUT_HOURS", "6"))
SCHEDULE = os.environ.get("DBT_DATABRICKS_PIPELINE_SCHEDULE") or None  # cron string; None = triggered only
RETRIES = int(os.environ.get("DBT_TASK_RETRIES", "2" if C.IS_PROD else "0"))

GRAPH = C.load_graph()  # small JSON, validated (unique ids, known edges, acyclic)
NODES = {n["id"]: n for n in GRAPH["nodes"]}

if EXECUTION_MODE not in ("airflow_cli", "databricks_job"):
    raise ValueError(f"DBT_EXECUTION_MODE must be 'airflow_cli' or 'databricks_job', got {EXECUTION_MODE!r}")
JOB_CONFIG = None
if EXECUTION_MODE == "databricks_job":
    from include.Python.dbt_databricks_job_operator import DbtDatabricksJobOperator, load_job_config
    JOB_CONFIG = load_job_config()

PARAMS = {
    "dbt_command": Param("run", type="string", enum=["run", "build", "test"],
                         description="run = models only; build = models + tests; test = tests only"),
    "select_nodes": Param("", type="string", description="Comma list of task ids / domains to run (empty = all)"),
    "skip_nodes": Param("", type="string", description="Comma list of task ids / domains to skip"),
    "full_refresh": Param(False, type="boolean", description="dbt --full-refresh"),
    "threads": Param(int(os.environ.get("DBT_THREADS", "4")), type="integer", minimum=1, maximum=64),
    "exclude": Param("", type="string", description="Additional dbt --exclude selector"),
    "dbt_vars": Param("", type="string", description="dbt --vars as YAML/JSON, e.g. {run_date: 2026-01-01}"),
    "fail_fast": Param(False, type="boolean"),
    "use_dbt_retry": Param(True, type="boolean", description="On Airflow retry run `dbt retry` (failed/skipped nodes only)"),
    "skip_preflight": Param(False, type="boolean", description="Bypass pre-flight checks (not recommended)"),
    "strict_graph_check": Param(True, type="boolean", description="Fail (not just warn) when the committed graph is stale"),
}


def _make_node_task(node: dict):
    common = dict(
        task_id=node["task"],
        trigger_rule=TriggerRule.NONE_FAILED,  # an excluded/skipped upstream must not block downstream
        execution_timeout=timedelta(hours=NODE_TIMEOUT_HOURS),
        doc_md=(
            f"**{node['id']}**: {node['models']} models, {node['tests']} tests\n\n"
            f"selector: `{node['selector']}`\n\nsources: {', '.join(node['sources']) or '-'}"
            + ("\n\n_Merged because these slices depend on each other in both directions._" if node["merged_cycle"] else "")
        ),
    )
    if EXECUTION_MODE == "databricks_job":
        return DbtDatabricksJobOperator(
            dbt_node_id=node["id"], dbt_domain=node["group"], select=node["selector"], job_config=JOB_CONFIG, **common
        )
    return DbtDatabricksBashOperator(select=node["selector"], dbt_node_id=node["id"], dbt_domain=node["group"], **common)


with DAG(
    dag_id="DBT_DATABRICKS_PIPELINE",
    description="Dependency-ordered dbt run of the dbt_databricks project on Databricks",
    doc_md=__doc__,
    start_date=pendulum.datetime(2025, 9, 1, tz="America/Chicago"),
    schedule=SCHEDULE,
    catchup=False,
    max_active_runs=1,  # two full runs would race on the same incremental tables
    max_active_tasks=MAX_PARALLEL_NODES,
    params=PARAMS,
    default_args={"owner": "data-platform", "retries": RETRIES, "retry_delay": timedelta(minutes=5)},
    tags=["dbt", "databricks", "dbt_databricks"],
) as dag:

    @task(task_id="preflight", retries=0)
    def preflight(**context):
        from include.Python.dbt_databricks_preflight import run_preflight
        return run_preflight(dict(context["params"]), check_graph_file=True)

    start = preflight()
    end = EmptyOperator(task_id="pipeline_complete", trigger_rule=TriggerRule.NONE_FAILED)

    groups: dict[str, TaskGroup] = {}
    tasks: dict[str, object] = {}
    for node in GRAPH["nodes"]:
        if node["group"] not in groups:  # (setdefault would build a TaskGroup on every iteration)
            groups[node["group"]] = TaskGroup(group_id=node["group"])
        with groups[node["group"]]:
            tasks[node["id"]] = _make_node_task(node)

    has_upstream = {down for _, down in GRAPH["edges"]}
    has_downstream = {up for up, _ in GRAPH["edges"]}
    for up, down in GRAPH["edges"]:
        tasks[up] >> tasks[down]
    for node_id, t in tasks.items():
        if node_id not in has_upstream:
            start >> t
        if node_id not in has_downstream:
            t >> end

"""
### DBT_DATABRICKS_EXECUTION

Run **any dbt selector** against the `dbt_databricks` project on Databricks - the Databricks
counterpart of `DBT_EXECUTION_NEW_PLATFORM`. Use it for ad-hoc runs, or let a driver / ingestion DAG
trigger it after new data lands.

Trigger with config, for example:

    {"select": "source:SUPPLIER+"}                       # everything downstream of one source
    {"select": "+tag:finance", "dbt_command": "build"}   # a tag, with its parents, models + tests
    {"select": "path:models/tea_service/bronze"}         # one layer of one domain
    {"model_name": "BRONZE__TEH_DATA_OPS__KANBANIZE_CARDS"}   # same key the existing DRIVER_* DAGs send

`select` is a normal dbt selector (space separated = union). `model_name` is accepted as an alias.

"""
### DATABRICKS_SOURCE_INGESTION

Loads **one source system into the Databricks landing zone**, then (optionally) refreshes everything in
dbt that depends on it.

    plan -> check_landing -> extract_load (one mapped task per table) -> validate_load -> trigger dbt

* **Source -> Databricks**: the existing extractors do the work, unchanged -
  `TelTools_SQLServer.main` / `TelTools_Oracle.main` read the source in chunks and write to the landing
  table via `databricks_teltools` (parquet -> Unity Catalog Volume -> `INSERT ... SELECT`), recording each
  run in `<LZ catalog>.<SCHEMA>.CONTROLTABLE`.
* **check_landing** fails in seconds if a landing table is missing (the loader never auto-creates tables).
* **validate_load** confirms every table wrote a CONTROLTABLE row during this run and logs row counts.
* **trigger dbt** runs `DBT_DATABRICKS_EXECUTION` with `source:<name>+` for the dbt sources this config
  feeds (mapping generated into `dbt_databricks_graph.json`), so bronze -> silver -> gold refresh in order.

Trigger with config, for example:

    {"config_filename": "TelTools_SQLServer_SUPPLIER_Extract.json"}
    {"config_filename": "TelTools_Oracle_Wisdom_Extract.json", "tables": "MAC_MAIN,SAP_SURB", "run_dbt": false}

| param | meaning |
|---|---|
| `config_filename` | an extract config in `include/Configuration` (SQL Server or Oracle) |
| `tables` | comma list of *source* table names; empty = every table in the config |
| `run_dbt` | trigger the dependent dbt run afterwards (default true) |
| `dbt_command` | `run` (default) / `build` / `test` for that dbt run |
| `check_landing_tables` | verify landing tables exist first (default true) |
| `fail_on_empty_load` | fail if any table loaded 0 rows (default false: normal for incremental loads) |

Sources not covered here (SAP R3 via `getSapR3Data`, HVR replication, file/API pulls, ...) keep their
existing DAGs; `DBT_DATABRICKS_PIPELINE` picks their landed data up.

from datetime import timedelta

import pendulum

from include.Python import dbt_databricks_common as C
from include.Python import databricks_ingestion as ING
from include.Python.dbt_databricks_airflow_compat import (
    DAG, Param, TriggerDagRunOperator, TriggerRule, get_current_context, task,

MAX_PARALLEL_TABLES = int(os.environ.get("INGEST_MAX_PARALLEL_TABLES", "6"))

CONFIG_CHOICES = [""] + ING.discover_config_files()

PARAMS = {

                             description="Extract config to load (TelTools_SQLServer_*_Extract.json / TelTools_Oracle_*_Extract.json)"),
    "tables": Param("", type="string", description="Comma list of source tables; empty = all in the config"),
    "run_dbt": Param(True, type="boolean", description="Trigger the dependent dbt run after loading"),
    "dbt_command": Param("run", type="string", enum=["run", "build", "test"]),
    "check_landing_tables": Param(True, type="boolean"),
    "fail_on_empty_load": Param(False, type="boolean"),
}

with DAG(
    dag_id="DATABRICKS_SOURCE_INGESTION",
    description="Extract one source system and load it into the Databricks landing zone, then refresh dependent dbt models",
    doc_md=__doc__,
    start_date=pendulum.datetime(2025, 9, 1, tz="America/Chicago"),
    schedule=None,
    catchup=False,
    max_active_runs=int(os.environ.get("INGEST_MAX_ACTIVE_RUNS", "6")),

    tags=["databricks", "ingestion", "teltools"],
) as dag:

    @task(task_id="plan", retries=0)
    def plan(**context):
        p = context["params"]
        result = ING.plan_ingestion(p.get("config_filename", ""), p.get("tables", ""))
        print(f"{result['config_filename']}: {len(result['tables'])} table(s) -> "
              f"{result['landing_schema']} schema; dbt selector: {result['dbt_select'] or '(none)'}")
        return result

        return plan_result["tables"]  # dynamic task mapping needs a list-valued XCom, not a dict key

"""
### DATABRICKS_END_TO_END

The **full workflow in one DAG**: source systems -> Databricks landing -> dbt bronze/silver/gold.

    check_databricks -> plan_ingestion -> ingest_source (one DATABRICKS_SOURCE_INGESTION run per source,
    in parallel) -> dbt_wanted -> run_dbt_pipeline (DBT_DATABRICKS_PIPELINE, dependency ordered)

* Every source ingestion run is a separate `DATABRICKS_SOURCE_INGESTION` DAG run (visible, retryable on
  its own). They are triggered with `run_dbt=false`; dbt runs **once** at the end over the whole graph so
  models that combine several sources see all of them freshly loaded.
* By default a failed ingestion stops the dbt stage (data would be inconsistent). Set the deployment
  variable `E2E_CONTINUE_ON_INGESTION_FAILURE=true` to run dbt anyway, or exclude the failing source with
  the `sources` param / `skip_ingestion`.
* Sources that are not extracted by Airflow's SQL Server/Oracle extractors (SAP R3, HVR, files, APIs) keep
  their own DAGs; run them first or on their own schedule - dbt simply reads whatever has landed.

Schedule: none by default. Set `DATABRICKS_E2E_SCHEDULE` (cron, e.g. `0 2 * * *`) on the deployment.

| param | meaning |
|---|---|
| `sources` | comma list of extractor names (`SUPPLIER`, `Wisdom`, `sqlserver:QVICS`, or a config file name); empty = all |
| `skip_ingestion` | only run dbt |
| `skip_dbt` | only ingest |

        """Turn the `sources` param into one DATABRICKS_SOURCE_INGESTION conf per extract config."""
        p = context["params"]
        if p.get("skip_ingestion"):
            print("skip_ingestion=true: nothing to ingest")

# --------------------------------------------------------------------------- graph
@lru_cache(maxsize=1)
def load_graph(path: str = GRAPH_PATH) -> dict[str, Any]:
    """Load and validate the pre-computed dbt dependency graph."""
    try:
        with open(path, "r", encoding="utf-8") as fh:
            graph = json.load(fh)
    except FileNotFoundError as exc:
        raise FileNotFoundError(
            f"dbt graph file not found: {path}. Generate it with "
            f"`python include/Python/generate_dbt_databricks_graph.py --parse` and deploy it with the DAGs."
        ) from exc

    if graph.get("schema_version") != 1:
        raise ValueError(f"Unsupported graph schema_version: {graph.get('schema_version')}")

    ids = [n["id"] for n in graph["nodes"]]
    if len(ids) != len(set(ids)):
        raise ValueError("Duplicate node ids in dbt graph file")
    known = set(ids)
    for up, down in graph["edges"]:
        if up not in known or down not in known:
            raise ValueError(f"Edge references unknown node: {up} -> {down}")
    assert_acyclic(ids, graph["edges"])
    return graph


def assert_acyclic(ids: Iterable[str], edges: Iterable[Iterable[str]]) -> None:
    """Kahn's algorithm; raises ValueError naming the nodes stuck in a cycle."""
    ids = list(ids)
    indeg = {i: 0 for i in ids}
    adj: dict[str, list[str]] = {i: [] for i in ids}
    for up, down in edges:
        adj[up].append(down)
        indeg[down] += 1
    ready = [i for i, d in indeg.items() if d == 0]
    seen = 0
    while ready:
        cur = ready.pop()
        seen += 1
        for nxt in adj[cur]:
            indeg[nxt] -= 1
            if indeg[nxt] == 0:
                ready.append(nxt)
    if seen != len(ids):
        stuck = sorted(i for i, d in indeg.items() if d > 0)
        raise ValueError(f"Cycle in dbt task graph involving: {stuck[:10]}")


# --------------------------------------------------------------------------- fingerprints & static checks
_REF_RE = re.compile(r"""\bref\(\s*(?:['"][^'"]+['"]\s*,\s*)?['"]([^'"]+)['"]""")
_SRC_RE = re.compile(r"""\bsource\(\s*['"]([^'"]+)['"]\s*,\s*['"]([^'"]+)['"]""")
_MAT_RE = re.compile(r"""materialized\s*=\s*['"]([A-Za-z0-9_]+)['"]""")
_MATERIALIZATION_DEF_RE = re.compile(r"""\{%-?\s*materialization\s+([A-Za-z0-9_]+)\s*,""")


def _model_files(project_dir: str) -> list[Path]:
    root = Path(project_dir) / "models"
    return sorted(p for p in root.rglob("*") if p.is_file() and p.suffix.lower() == ".sql")


def dependency_fingerprint(project_dir: str = DBT_PROJECT_DIR) -> str:
    """
    Hash of every model file path plus the ref()/source() calls inside it.

    This is what the dependency graph is derived from, so if it changes the
    committed graph JSON is stale. Whitespace/formatting edits and ordinary SQL
    changes do NOT change it. (Refs produced by macros are not visible here;
    dbt still orders models correctly inside a task, only cross-task ordering
    could be affected.)
    """
    h = hashlib.sha1()
    base = Path(project_dir)
    for f in _model_files(project_dir):
        text = f.read_text(encoding="utf-8", errors="ignore")
        refs = sorted(set(_REF_RE.findall(text)))
        srcs = sorted({f"{a}.{b}" for a, b in _SRC_RE.findall(text)})
        h.update(f"{f.relative_to(base).as_posix().lower()}|{','.join(refs)}|{','.join(srcs)}\n".encode())
    return h.hexdigest()


def unresolved_materializations(project_dir: str = DBT_PROJECT_DIR) -> dict[str, int]:
    """
    Materializations named in model configs that are neither dbt built-ins nor
    defined as a `{% materialization name, adapter='...' %}` macro in the
    project or its installed packages. Returns {name: model_count}.
    """
    defined = set(BUILTIN_MATERIALIZATIONS)
    for sub in ("macros", "dbt_packages", "dbt_internal_packages"):
        root = Path(project_dir) / sub
        if root.is_dir():
            for f in root.rglob("*.sql"):
                try:
                    defined.update(_MATERIALIZATION_DEF_RE.findall(f.read_text(encoding="utf-8", errors="ignore")))
                except OSError:
                    continue
    counts: dict[str, int] = {}
    for f in _model_files(project_dir):
        for name in set(_MAT_RE.findall(f.read_text(encoding="utf-8", errors="ignore"))):
            if name not in defined:
                counts[name] = counts.get(name, 0) + 1
    return counts

def missing_dbt_env(environ: dict[str, str] | None = None, allow_connection: bool = True) -> list[str]:
    environ = os.environ if environ is None else environ
    missing = [k for k in REQUIRED_DBT_ENV if not environ.get(k)]
    if allow_connection:
        missing = [k for k in missing if k not in CONNECTION_BACKED_ENV]
        if any(not environ.get(k) for k in CONNECTION_BACKED_ENV):
            # verify the connection can actually supply them
            try:
                supplied = env_from_connection()
            except Exception:  # noqa: BLE001 - connection absent is a normal "missing" case
                supplied = {}
            missing += [k for k in CONNECTION_BACKED_ENV if not environ.get(k) and not supplied.get(k)]
    return missing


def env_from_connection(conn_id: str = DATABRICKS_CONN_ID) -> dict[str, str]:
    """
    DATABRICKS_HOST / HTTP_PATH / TOKEN derived from an Airflow Databricks
    connection (host, extra.http_path, password or extra.token). Only called
    at task run time.
    """
    try:
        from airflow.sdk import Connection  # Airflow 3
        conn = Connection.get(conn_id)
    except ImportError:  # Airflow 2
        from airflow.hooks.base import BaseHook
        conn = BaseHook.get_connection(conn_id)

    extra = conn.extra_dejson if hasattr(conn, "extra_dejson") else json.loads(conn.extra or "{}")
    host = (conn.host or "").replace("https://", "").replace("http://", "").strip("/")
    out = {
        "DATABRICKS_HOST": host,
        "DATABRICKS_HTTP_PATH": extra.get("http_path", ""),
        "DATABRICKS_TOKEN": conn.password or extra.get("token", ""),
    }
    return {k: v for k, v in out.items() if v}


def validate_selector(value: str, label: str = "selector") -> str:
    value = (value or "").strip()
    if not _SELECTOR_OK.match(value):
        raise ValueError(f"Unsupported characters in dbt {label}: {value!r}")
    return value


def run_key(*parts: str) -> str:
    """Filesystem-safe key for per-task dbt target/log directories."""
    return "/".join(re.sub(r"[^A-Za-z0-9_.-]+", "_", p or "x") for p in parts)

Operators for running the dbt_databricks project from Airflow.

DbtDatabricksBashOperator
    Default execution mode. Runs dbt from the `dbt_venv` on the Airflow worker;
    dbt sends SQL to the Databricks SQL warehouse (as described in the dbt
    project README).

build_databricks_job_operator()
    Optional execution mode (DBT_EXECUTION_MODE=databricks_job). Submits the same
    dbt command as a Databricks Jobs one-time run with a `dbt_task`.
"""
from __future__ import annotations

import json
import os
from typing import Any

try:  # Airflow 3 / standard provider
    from airflow.providers.standard.operators.bash import BashOperator
except ImportError:  # Airflow 2
    from airflow.operators.bash import BashOperator

try:
    from airflow.exceptions import AirflowSkipException
except ImportError:  # pragma: no cover
    from airflow.sdk.exceptions import AirflowSkipException

ALLOWED_COMMANDS = ("run", "build", "test")


def csv_set(value: Any) -> set[str]:
    """'a, b,,c' / ['a','b'] -> {'a','b','c'} (lower-cased)."""
    if not value:
        return set()
    if isinstance(value, str):
        value = value.split(",")
    return {str(v).strip().lower() for v in value if str(v).strip()}


def node_is_selected(node_id: str, domain: str, params: dict[str, Any]) -> bool:
    """`select_nodes` (empty = all) and `skip_nodes` match on node id or domain."""
    keys = {node_id.lower(), domain.lower()}
    selected, skipped = csv_set(params.get("select_nodes")), csv_set(params.get("skip_nodes"))
    if skipped & keys:
        return False
    return not selected or bool(selected & keys)


def bool_param(value: Any) -> bool:
    return str(value).strip().lower() in ("true", "1", "yes", "y")


class DbtDatabricksBashOperator(BashOperator):
    """
    One dbt invocation (run / build / test) for a fixed or parameter-supplied selector.

    dbt_node_id / dbt_domain : set for pipeline tasks; used by the `select_nodes` / `skip_nodes` params.
    select           : fixed dbt selector (space separated), e.g. "path:models/tea_service/bronze".
    select_param     : alternative: names of DAG params/conf keys to read the selector from.
    """

    ui_color = "#ffb199"
    template_fields = ("select",)  # bash_command is a constant and is intentionally NOT templated

    def __init__(
        self,
        *,
        select: str | None = None,
        select_param: tuple[str, ...] = (),
        dbt_node_id: str | None = None,
        dbt_domain: str | None = None,
        **kwargs: Any,
    ) -> None:
        # Placeholder keeps BashOperator happy; the real script is set in execute()
        # (after templating) so Jinja never sees the shell code.
        kwargs.setdefault("do_xcom_push", False)  # don't store dbt's last stdout line as an XCom
        super().__init__(bash_command="true", append_env=True, **kwargs)
        self.select = select
        self.select_param = select_param
        self.dbt_node_id = dbt_node_id
        self.dbt_domain = dbt_domain

    def execute(self, context: Any) -> Any:
        params = dict(context["params"])
        ti = context["ti"]
        # Existing DRIVER_* DAGs trigger with {"model_name": ...}; honour conf keys that are not declared params.
        conf = getattr(context.get("dag_run"), "conf", None) or {}
        for key in ("select", "model_name"):
            if not params.get(key) and conf.get(key):
                params[key] = conf[key]

        if self.dbt_node_id and not node_is_selected(self.dbt_node_id, self.dbt_domain or "", params):
            raise AirflowSkipException(f"{self.dbt_node_id} excluded by select_nodes/skip_nodes")

        command = str(params.get("dbt_command") or "run").lower()
        if command not in ALLOWED_COMMANDS:
            raise ValueError(f"dbt_command must be one of {ALLOWED_COMMANDS}, got {command!r}")

        select = self.select
        if not select:
            for key in self.select_param:
                if params.get(key):
                    select = str(params[key])
                    break
        if not select:
            raise ValueError(f"No dbt selector supplied (looked at: {self.select_param or 'static select'})")
        select = C.validate_selector(select, "select")
        exclude = C.validate_selector(str(params.get("exclude") or ""), "exclude")
        dbt_vars = str(params.get("dbt_vars") or "").strip()

        profiles, target = C.profiles_dir()
        env: dict[str, str] = {
            "DBT_VENV_ACTIVATE": C.DBT_VENV_ACTIVATE,
            "DBT_PROJECT_DIR": C.DBT_PROJECT_DIR,
            "TEL_DEPLOYMENT": C.DEPLOYMENT,
            "TEL_PROFILES_DIR": profiles,
            "TEL_DBT_TARGET": target or "",
            "TEL_RUN_KEY": C.run_key(context["dag"].dag_id, context["run_id"], ti.task_id),
            "TEL_TRY_NUMBER": str(getattr(ti, "try_number", 1) or 1),
            "TEL_DBT_COMMAND": command,
            "TEL_DBT_SELECT": select,
            "TEL_DBT_EXCLUDE": exclude,
            "TEL_DBT_THREADS": str(int(params.get("threads") or 4)),
            "TEL_DBT_FULL_REFRESH": "true" if bool_param(params.get("full_refresh")) else "false",
            "TEL_DBT_FAIL_FAST": "true" if bool_param(params.get("fail_fast")) else "false",
            "TEL_DBT_VARS": dbt_vars,
            "TEL_USE_DBT_RETRY": "true" if bool_param(params.get("use_dbt_retry", True)) else "false",
        }

        # Fill any DATABRICKS_HOST / HTTP_PATH / TOKEN not already set on the worker
        # from the Airflow connection. Existing env vars always win.
        if any(not os.environ.get(k) for k in C.CONNECTION_BACKED_ENV):
            for key, value in C.env_from_connection().items():
                if not os.environ.get(key):
                    env[key] = value

        self.env = env
        self.bash_command = C.DBT_SHELL_SCRIPT
        return super().execute(context)

"""
Optional execution mode: run each dbt slice as a Databricks Jobs one-time run
with a `dbt_task` (Jobs API 2.1 `runs/submit`), instead of running dbt on the
Airflow worker.

Enable with DBT_EXECUTION_MODE=databricks_job on the Airflow deployment and
provide the job settings in include/Configuration/dbt_databricks_job.json (or the
DBT_DATABRICKS_JOB_CONFIG env var holding the same JSON). See
dbt_databricks_job.example.json for every key.

IMPORTANT - what this mode needs on the Databricks side
  * The dbt project must be reachable by the job through a Git source
    (`git_source`); Databricks cannot run a dbt project from DBFS.
  * The project reads its catalogs/credentials with env_var(). A job cluster
    therefore needs them as `spark_env_vars`. Non-secret DATABRICKS_* values
    present on the Airflow deployment are forwarded automatically; secrets
    (DATABRICKS_TOKEN, ...) must be given as secret references such as
    "{{secrets/<scope>/<key>}}" under `secret_env_vars` in the job config.
  * Use a custom profile (`profiles_directory`, e.g. PROD_PROFILES) so the
    project's own profile - which uses those env vars - is what dbt reads.

This mode is NOT exercised by the default setup and has not been run against a
live workspace; validate it in DEV first. The default `airflow_cli` mode follows
the dbt project README and is the supported path.
"""
from __future__ import annotations

import json
import os
from pathlib import Path
from typing import Any

from airflow.exceptions import AirflowSkipException
from airflow.providers.databricks.operators.databricks import DatabricksSubmitRunOperator

from include.Python import dbt_databricks_common as C
from include.Python.dbt_databricks_operators import node_is_selected

JOB_CONFIG_PATH = os.environ.get(
    "DBT_DATABRICKS_JOB_CONFIG_PATH", str(Path(C.CONFIG_DIR) / "dbt_databricks_job.json")
)
# Never forwarded as plain spark_env_vars (per-user / service-principal secrets).
_NEVER_FORWARD = {"DATABRICKS_TOKEN", "DATABRICKS_DEV_CLIENT", "DATABRICKS_DEV_SECRET"}


def load_job_config() -> dict[str, Any]:
    raw = os.environ.get("DBT_DATABRICKS_JOB_CONFIG")
    if raw:
        cfg = json.loads(raw)
    else:
        try:
            cfg = json.loads(Path(JOB_CONFIG_PATH).read_text(encoding="utf-8"))
        except FileNotFoundError as exc:
            raise FileNotFoundError(
                f"DBT_EXECUTION_MODE=databricks_job needs a job config: {JOB_CONFIG_PATH} "
                f"(or env DBT_DATABRICKS_JOB_CONFIG). See dbt_databricks_job.example.json."
            ) from exc
    for key in ("git_source", "project_directory", "job_cluster"):
        if not cfg.get(key):
            raise ValueError(f"dbt_databricks_job config is missing required key: {key}")
    return cfg


def _spark_env_vars(cfg: dict[str, Any]) -> dict[str, str]:
    env = {
        k: v for k, v in os.environ.items()
        if k.startswith(("DATABRICKS_", "DBT_THREADS", "DBT_UPLOAD_ARTIFACTS")) and k not in _NEVER_FORWARD and v
    }
    env.update(cfg.get("spark_env_vars", {}))
    env.update(cfg.get("secret_env_vars", {}))
    env["ASTRO_DEPLOYMENT_NAME"] = C.DEPLOYMENT
    return env


def _raw(value: Any) -> Any:
    """Protect config-derived strings from Jinja.

    The operator's `json` field is templated, but Databricks secret references look like
    `{{secrets/<scope>/<key>}}` - the same braces Jinja uses. Without this they would fail to render
    (or silently become empty). Only the dbt command line is meant to be templated.
    """
    if isinstance(value, str):
        return "{% raw %}" + value + "{% endraw %}" if ("{{" in value or "{%" in value or "{#" in value) else value
    if isinstance(value, dict):
        return {k: _raw(v) for k, v in value.items()}
    if isinstance(value, list):
        return [_raw(v) for v in value]
    return value


class DbtDatabricksJobOperator(DatabricksSubmitRunOperator):
    """DatabricksSubmitRunOperator that honours select_nodes/skip_nodes and builds the dbt_task payload."""

    def __init__(self, *, dbt_node_id: str | None, dbt_domain: str | None, select: str, job_config: dict[str, Any], **kwargs: Any):
        C.validate_selector(select, "select")
        cfg = job_config
        # profiles_directory in the job is repo-relative: <project_directory>/<DEPLOYMENT>_PROFILES
        profiles_directory = cfg.get("profiles_directory") or f"{cfg['project_directory'].rstrip('/')}/{C.DEPLOYMENT}_PROFILES"

        # dbt commands are templated so DAG params (dbt_command, threads, full_refresh) apply per run.
        run_cmd = (
            "dbt {{ params.dbt_command }} --select " + select +
            " --threads {{ params.threads }}"
            "{% if params.full_refresh %} --full-refresh{% endif %}"
            "{% if params.fail_fast %} --fail-fast{% endif %}"
            "{% if params.exclude %} --exclude {{ params.exclude }}{% endif %}"
        )
        dbt_task: dict[str, Any] = {
            "project_directory": cfg["project_directory"],
            "profiles_directory": profiles_directory,
            "commands": ["dbt deps", run_cmd],
            "source": "GIT",
        }
        if cfg.get("warehouse_id"):
            dbt_task["warehouse_id"] = cfg["warehouse_id"]

        cluster = dict(cfg["job_cluster"])
        cluster["spark_env_vars"] = {**cluster.get("spark_env_vars", {}), **_spark_env_vars(cfg)}
        cluster = _raw(cluster)

        payload = {
            "run_name": f"airflow-dbt-{dbt_node_id or 'adhoc'}",
            "git_source": _raw(cfg["git_source"]),
            "timeout_seconds": int(cfg.get("timeout_seconds", 21600)),
            "tasks": [
                {
                    "task_key": (dbt_node_id or "adhoc").replace(".", "_")[:100],
                    "dbt_task": dbt_task,
                    "new_cluster": cluster,
                    "libraries": [{"pypi": {"package": cfg.get("dbt_pip_spec", "dbt-databricks>=1.7.17,<1.8")}}],
                }
            ],
        }
        super().__init__(
            json=payload,
            databricks_conn_id=C.DATABRICKS_CONN_ID,
            polling_period_seconds=int(cfg.get("polling_period_seconds", 30)),
            databricks_retry_limit=3,
            **kwargs,
        )
        self.dbt_node_id = dbt_node_id
        self.dbt_domain = dbt_domain

    def pre_execute(self, context: Any) -> None:
        if self.dbt_node_id and not node_is_selected(self.dbt_node_id, self.dbt_domain or "", dict(context["params"])):
            raise AirflowSkipException(f"{self.dbt_node_id} excluded by select_nodes/skip_nodes")

"""
Pre-flight checks run as the first task of the dbt DAGs.

Purpose: fail in seconds with an actionable message instead of letting ~100
dbt tasks fail one by one. Each check returns (level, message); level is

def _truthy(v: Any) -> bool:

def run_in_dbt_venv(cmd: str, extra_env: dict[str, str] | None = None, timeout: int = 180) -> subprocess.CompletedProcess:
    """Run a command with the dbt venv activated, inside the dbt project directory."""
    env = {**os.environ, **(extra_env or {})}
    env["DBT_VENV_ACTIVATE"] = C.DBT_VENV_ACTIVATE
    env["DBT_PROJECT_DIR"] = C.DBT_PROJECT_DIR
    return subprocess.run(
        ["bash", "-c", f'source "$DBT_VENV_ACTIVATE" && cd "$DBT_PROJECT_DIR" && {cmd}'],
        env=env, capture_output=True, text=True, timeout=timeout,
    )


def check_project() -> list[Finding]:
    out: list[Finding] = []
    proj = Path(C.DBT_PROJECT_DIR)
    if not (proj / "dbt_project.yml").is_file():
        return [("error", f"dbt project not found at {proj} (set DBT_DATABRICKS_PROJECT_DIR)")]
    pdir, _ = C.profiles_dir()
    if not (Path(pdir) / "profiles.yml").is_file():
        out.append(("error", f"profiles.yml not found in {pdir}"))
    if not Path(C.DBT_VENV_ACTIVATE).is_file():
        out.append(("error", f"dbt virtualenv not found: {C.DBT_VENV_ACTIVATE} (see Dockerfile: dbt_venv)"))
    return out

    text = (res.stdout or "") + (res.stderr or "")

def run_preflight(params: dict[str, Any], *, check_graph_file: bool = True) -> dict[str, Any]:
    """Run all checks; raise on errors. Returns a small summary for XCom."""
    if _truthy(params.get("skip_preflight")):
        log.warning("skip_preflight=true: pre-flight checks were NOT run")
        return {"skipped": True}

    findings: list[Finding] = []
    findings += check_project()
    if not any(level == "error" for level, _ in findings):
        findings += check_project_name()
        findings += check_env()
        if check_graph_file:
            findings += check_graph(strict=_truthy(params.get("strict_graph_check", True)))
        findings += check_materializations()
    if not any(level == "error" for level, _ in findings):
        findings += check_connection()  # only worth the round trip if everything else is sane

    for level, msg in findings:
        (log.error if level == "error" else log.warning)("[preflight:%s] %s", level, msg)
    errors = [m for level, m in findings if level == "error"]
    if errors:
        from airflow.exceptions import AirflowFailException
        raise AirflowFailException(
            f"Pre-flight failed with {len(errors)} problem(s). Fix them, or re-run with skip_preflight=true "
            f"to bypass (not recommended).\n\n" + "\n\n".join(f"- {m}" for m in errors)
        )
    return {"skipped": False, "warnings": [m for level, m in findings if level == "warn"]}

"""
Source-system -> Databricks landing helpers used by DATABRICKS_SOURCE_INGESTION.

The extraction + load itself is done by the EXISTING extractors, unchanged:

    include/Python/TelTools_SQLServer.py   main(config_file, table)
    include/Python/TelTools_Oracle.py      main(config_file, table)

Those read the source (SQL Server / Oracle), page it in chunks and write it to the
landing table through TELTools.putSnowflakeDataFrame, which routes to
include/Python/databricks_teltools.py (parquet -> UC Volume -> INSERT ... SELECT)
and records each run in <LZ catalog>.<SCHEMA>.CONTROLTABLE.

What this module adds around them:
    plan_ingestion      which tables a config file loads, and which dbt selector depends on them
    check_landing       fail fast if the landing schema / tables / CONTROLTABLE do not exist
                        (the loader never auto-creates tables)
    extract_one         the call into the right extractor
    validate_load       confirm every table wrote a CONTROLTABLE row during this run

Connection settings for the load are the existing ones (DATABRICKS_DEV_HOST,
DATABRICKS_DEV_HTTP_PATH, DATABRICKS_DEV_CLIENT, DATABRICKS_DEV_SECRET, and

    if not value:

    if isinstance(value, str):

def plan_ingestion(config_filename: str, tables: Any = "") -> dict[str, Any]:
    """Resolve what one ingestion run will load. JSON-serialisable (goes through XCom)."""
    config_filename = os.path.basename(config_filename or "")
    if not config_filename:
        raise ValueError("config_filename is required")
    kind, server = classify(config_filename)
    path = Path(C.CONFIG_DIR) / config_filename
    if not path.is_file():
        raise FileNotFoundError(f"Extract config not found: {path}")

    data = json.loads(path.read_text(encoding="utf-8-sig"))
    datasets = [
        {"table": v["table"], "landing_table": v["sftablename"]}
        for ds in data.values() if isinstance(ds, dict)
        for k, v in ds.items() if k.startswith("DataSet") and isinstance(v, dict)
    ]
    requested = _csv(tables)
    if requested:
        known = {d["table"].lower() for d in datasets}
        unknown = [t for t in requested if t.lower() not in known]
        if unknown:
            raise ValueError(f"Tables not in {config_filename}: {unknown}")
        wanted = {t.lower() for t in requested}
        datasets = [d for d in datasets if d["table"].lower() in wanted]
    if not datasets:
        raise ValueError(f"No datasets found in {config_filename}")

    # same derivation the extractors use for the landing schema
    landing_schema = server.upper().split("-")[0] if kind == "sqlserver" else server.upper()

    dbt_select = ""
    try:
        for ex in C.load_graph().get("extractors", []):
            if ex["config_filename"] == config_filename:
                dbt_select = ex.get("dbt_select", "")
                break
    except Exception as exc:  # noqa: BLE001 - graph is optional for ingestion
        log.warning("dbt graph unavailable, downstream dbt selector left empty: %s", exc)

    return {
        "config_filename": config_filename,
        "kind": kind,
        "server": server,
        "landing_schema": landing_schema,
        "datasets": datasets,
        "tables": [d["table"] for d in datasets],
        "dbt_select": dbt_select,
        "run_start": dt.datetime.now().strftime("%Y-%m-%d %H:%M:%S.%f"),  # same clock/format the extractors stamp RUN_TIME with
    }

def check_landing(plan: dict[str, Any]) -> dict[str, Any]:
    """Verify the landing schema, every target table and CONTROLTABLE exist before extracting anything."""
    _require_loader_env()
    D, conn = _landing_connection(plan["landing_schema"])
    try:
        schema_fq = f"`{D._catalog(os.environ['SNOWFLAKE_LZ'])}`.`{D._ident(plan['landing_schema'])}`"
        with conn.cursor() as cur:
            cur.execute(f"SHOW TABLES IN {schema_fq}")
            present = {str(r[1]).lower() for r in cur.fetchall()}  # (database, tableName, isTemporary)
    finally:
        conn.close()

    expected = {d["landing_table"].lower() for d in plan["datasets"]} | {"controltable"}
    missing = sorted(expected - present)
    if missing:
        raise RuntimeError(
            f"Landing tables missing in {schema_fq}: {missing}. The loader never auto-creates landing tables - "
            f"provision them with the DDL generator (include/Python/createLandingZone.py / lz DDL) first."
        )
    log.info("Landing check OK: %d tables present in %s", len(expected), schema_fq)
    return {"schema": schema_fq, "tables_checked": len(expected)}

    """Extract one source table and load it to Databricks via the existing extractor."""
    _require_loader_env()
    kind, _ = classify(config_filename)
    path = str(Path(C.CONFIG_DIR) / os.path.basename(config_filename))
    if kind == "sqlserver":
        import include.Python.TelTools_SQLServer as extractor
    else:
        import include.Python.TelTools_Oracle as extractor
    log.info("Extracting %s from %s", table, config_filename)
    extractor.main(path, table)
    log.info("Loaded %s", table)


def validate_load(plan: dict[str, Any], fail_on_empty: bool = False) -> list[dict[str, Any]]:
    """Every table must have written a CONTROLTABLE row during this run (RUN_TIME >= run_start)."""
    _require_loader_env()
    D, conn = _landing_connection(plan["landing_schema"])
    since = (dt.datetime.strptime(plan["run_start"], "%Y-%m-%d %H:%M:%S.%f") - dt.timedelta(minutes=10)).strftime("%Y-%m-%d %H:%M:%S")
    fq = D._fq(os.environ["SNOWFLAKE_LZ"], plan["landing_schema"], "CONTROLTABLE")
    try:
        with conn.cursor() as cur:
            cur.execute(
                f"SELECT TABLE_NAME, ROW_COUNT, RUN_TIME FROM {fq} "
                f"WHERE CAST(RUN_TIME AS TIMESTAMP) >= CAST(:since AS TIMESTAMP) "
                f"QUALIFY row_number() OVER (PARTITION BY TABLE_NAME ORDER BY RUN_TIME DESC) = 1",
                {"since": since},
            )
            rows = {str(r[0]).lower(): r for r in cur.fetchall()}
    finally:
        conn.close()

    report, missing, empty = [], [], []
    for t in plan["tables"]:
        row = rows.get(t.lower())
        if row is None:
            missing.append(t)
            continue
        n = int(row[1]) if row[1] not in (None, "") else None
        report.append({"table": t, "rows": n, "run_time": str(row[2])})
        if n == 0:
            empty.append(t)
        log.info("%-60s rows=%s run_time=%s", t, n, row[2])
    if missing:
        raise RuntimeError(f"No CONTROLTABLE entry for this run (load did not complete?): {missing}")
    if empty and fail_on_empty:
        raise RuntimeError(f"Loads returned 0 rows and fail_on_empty_load=true: {empty}")
    if empty:
        log.warning("%d table(s) loaded 0 rows (normal for incremental loads with no new data): %s", len(empty), empty[:10])
    return report

"""
Generate include/Configuration/dbt_databricks_graph.json from dbt's manifest.

WHY A GENERATED FILE
    The project has ~5,500 models and its manifest is ~170 MB; parsing it takes
    minutes. Airflow re-imports DAG files constantly, so the DAGs read this small
    pre-computed graph instead. Re-run this script (CI step or by hand) whenever
    models are added/removed or their ref()/source() calls change. The DAGs'
    pre-flight task compares a dependency fingerprint and fails loudly if the
    graph is stale.

WHAT IT BUILDS
    * one NODE per domain/layer folder  (models/<domain>/<layer>), e.g. tea_service/bronze
      plus one node per installed dbt package that project models depend on
      (dbt_artifacts)
    * EDGES between nodes from the model-level ref() graph
    * nodes that sit in a cycle at this coarser level (e.g. teh_finance
      silver <-> gold, where individual models are acyclic) are merged into one
      node, so the Airflow graph is always a DAG; dbt orders the models inside it
    * edges are transitively reduced (same ordering, far fewer arrows)
    * source -> nodes and extractor-config -> source maps used by the ingestion DAGs

USAGE
    python generate_dbt_databricks_graph.py --parse            # deps + parse + generate
    python generate_dbt_databricks_graph.py --manifest path/to/manifest.json

    --parse runs `dbt deps` and `dbt parse` with placeholder connection values

def group_of(model: dict, root_pkg: str) -> tuple[str, str, str]:
    """(domain, layer, selector path) for a model node."""
    if model["package_name"] != root_pkg:
        return model["package_name"], "package", f"package:{model['package_name']}"
    parts = model["original_file_path"].replace("\\", "/").split("/")   # models/<domain>/<layer>/file.sql
    if len(parts) >= 4:
        return parts[1], parts[2], f"path:{'/'.join(parts[:3])}"
    return parts[1] if len(parts) > 2 else "root", "_", f"path:{'/'.join(parts[:-1])}"


def tarjan_scc(nodes: list, adj: dict) -> list[list]:
    index, low, on, stack, out, counter = {}, {}, set(), [], [], [0]

    def visit(v):
        # iterative to stay safe on deep graphs
        work = [(v, iter(adj.get(v, ())))]
        index[v] = low[v] = counter[0]; counter[0] += 1
        stack.append(v); on.add(v)
        while work:
            node, it = work[-1]
            advanced = False
            for w in it:
                if w not in index:
                    index[w] = low[w] = counter[0]; counter[0] += 1
                    stack.append(w); on.add(w)
                    work.append((w, iter(adj.get(w, ()))))
                    advanced = True
                    break
                elif w in on:
                    low[node] = min(low[node], index[w])
            if advanced:
                continue
            work.pop()
            if work:
                low[work[-1][0]] = min(low[work[-1][0]], low[node])
            if low[node] == index[node]:
                comp = []
                while True:
                    w = stack.pop(); on.discard(w); comp.append(w)
                    if w == node:
                        break
                out.append(sorted(comp))

    for n in nodes:
        if n not in index:
            visit(n)
    return out


def transitive_reduction(ids: list[str], edges: set[tuple[str, str]]) -> set[tuple[str, str]]:
    adj = collections.defaultdict(set)
    for a, b in edges:
        adj[a].add(b)
    reach: dict[str, set[str]] = {}

    def descendants(n: str) -> set[str]:
        if n in reach:
            return reach[n]
        seen: set[str] = set()
        for m in adj[n]:
            seen.add(m)
            seen |= descendants(m)
        reach[n] = seen
        return seen

    sys.setrecursionlimit(10000)
    keep = set()
    for a, b in edges:
        # drop a->b if b is reachable from a through another direct successor
        if not any(b in descendants(m) for m in adj[a] if m != b):
            keep.add((a, b))
    return keep

def source_catalog_info(project_dir: Path) -> dict[str, dict]:
    """source name -> {catalog_env, schema_env, schema_default} read from the raw source YAMLs."""
    try:
        import yaml
    except ImportError:
        return {}
    info: dict[str, dict] = {}
    env_re = re.compile(r"env_var\(\s*'([A-Z0-9_]+)'(?:\s*,\s*'([^']*)')?")
    for f in sorted((project_dir / "models").rglob("*.yml")):
        try:
            doc = yaml.safe_load(f.read_text(encoding="utf-8")) or {}
        except Exception:  # noqa: BLE001
            continue
        for src in doc.get("sources", []) or []:
            name = src.get("name")
            if not name:
                continue
            cat = env_re.search(str(src.get("database", "")))
            sch = env_re.search(str(src.get("schema", "")))
            rec = info.setdefault(name, {"catalog_env": set(), "schema_env": set(), "schema_default": set()})
            if cat:
                rec["catalog_env"].add(cat.group(1))
            if sch:
                rec["schema_env"].add(sch.group(1))
                if sch.group(2):
                    rec["schema_default"].add(sch.group(2))
    return {k: {kk: sorted(vv) for kk, vv in v.items()} for k, v in info.items()}

def build(manifest_path: Path, project_dir: Path, config_dir: Path) -> dict:
    m = json.loads(manifest_path.read_text(encoding="utf-8"))
    root_pkg = m["metadata"].get("project_name") or project_dir.name
    nodes = m["nodes"]
    models = {k: v for k, v in nodes.items() if v["resource_type"] == "model"}
    tests = [v for v in nodes.values() if v["resource_type"] == "test"]

    grp = {k: group_of(v, root_pkg) for k, v in models.items()}
    keys = sorted(set(grp.values()))
    gid = lambda g: f"{g[0]}__{g[1]}"  # noqa: E731

    # model-level edges aggregated to (domain, layer) level
    edges: set[tuple[tuple, tuple]] = set()
    src_by_group: dict[tuple, set[str]] = collections.defaultdict(set)
    for k, v in models.items():
        for d in v["depends_on"]["nodes"]:
            if d in models:
                if grp[d] != grp[k]:
                    edges.add((grp[d], grp[k]))
            elif d.startswith("source."):
                src_by_group[grp[k]].add(m["sources"][d]["source_name"])

    # package models are only worth running if some project model needs them
    needed_pkgs = {a for a, b in edges if a[1] == "package"}
    keys = [g for g in keys if g[1] != "package" or g in needed_pkgs]
    edges = {(a, b) for a, b in edges if a in keys and b in keys}

    adj = collections.defaultdict(set)
    for a, b in edges:
        adj[a].add(b)
    comps = tarjan_scc(keys, adj)
    comp_of = {g: i for i, comp in enumerate(comps) for g in comp}

    counts = collections.Counter(grp[k] for k in models)
    test_counts: collections.Counter = collections.Counter()
    path_to_group = {}
    for g in keys:
        if g[1] != "package":
            path_to_group[f"models/{g[0]}/{g[1]}/"] = g
    for t in tests:
        p = t["original_file_path"].replace("\\", "/")
        for prefix, g in path_to_group.items():
            if p.startswith(prefix):
                test_counts[g] += 1
                break

    out_nodes = []
    node_name: dict[int, str] = {}
    for i, comp in enumerate(comps):
        domains = sorted({g[0] for g in comp})
        layers = [g[1] for g in comp]
        order = {"bronze": 0, "silver": 1, "gold": 2}
        comp_sorted = sorted(comp, key=lambda g: (order.get(g[1], 9), g[1]))
        if len(domains) == 1:
            domain = domains[0]
            nid = f"{domain}__" + "_".join(g[1] for g in comp_sorted)
            task = "_".join(g[1] for g in comp_sorted)
        else:
            domain = "merged"
            nid = "merged__" + "__".join(gid(g) for g in comp_sorted)
            task = "__".join(gid(g) for g in comp_sorted)
        node_name[i] = nid
        out_nodes.append({
            "id": nid,
            "group": domain,
            "task": task,
            "members": [gid(g) for g in comp_sorted],
            "selector": " ".join(g[2] for g in comp_sorted),
            "models": sum(counts[g] for g in comp),
            "tests": sum(test_counts[g] for g in comp),
            "sources": sorted(set().union(*[src_by_group[g] for g in comp])),
            "merged_cycle": len(comp) > 1,
        })

    node_edges = {(node_name[comp_of[a]], node_name[comp_of[b]]) for a, b in edges if comp_of[a] != comp_of[b]}
    ids = [n["id"] for n in out_nodes]
    reduced = transitive_reduction(ids, node_edges)
    C.assert_acyclic(ids, reduced)

    # sources
    cat_info = source_catalog_info(project_dir)
    nodes_by_source: dict[str, set[str]] = collections.defaultdict(set)
    for g, names in src_by_group.items():
        for s in names:
            nodes_by_source[s].add(node_name[comp_of[g]] if g in comp_of else gid(g))
    src_tables: dict[str, dict] = collections.defaultdict(lambda: {"schemas": set(), "tables": set()})
    for s in m["sources"].values():
        rec = src_tables[s["source_name"]]
        rec["schemas"].add(s["schema"])
        rec["tables"].add(s.get("identifier") or s["name"])
    sources = {}
    for name, rec in sorted(src_tables.items()):
        sources[name] = {
            "schemas": sorted(rec["schemas"]),
            "tables": sorted(rec["tables"]),
            "catalog_env": cat_info.get(name, {}).get("catalog_env", []),
            "nodes": sorted(nodes_by_source.get(name, [])),
        }

    # extractor configs -> dbt sources (match on landing schema + the LZ catalog)
    extractors = discover_extractors(config_dir) if config_dir.is_dir() else []
    for ex in extractors:
        matched = [
            n for n, s in sources.items()
            if any(sc.upper() == ex["landing_schema"] for sc in s["schemas"])
            and (not s["catalog_env"] or "DATABRICKS_LZ_CATALOG" in s["catalog_env"])
        ]
        ex["dbt_sources"] = sorted(matched)
        ex["dbt_select"] = " ".join(f"source:{n}+" for n in sorted(matched))
        dbt_tables = {t.lower() for n in matched for t in sources[n]["tables"]}
        ex["landing_tables_known_to_dbt"] = sum(1 for d in ex["datasets"] if d["landing_table"].lower() in dbt_tables)

    return {
        "schema_version": 1,
        "generated_at": dt.datetime.now(dt.timezone.utc).isoformat(timespec="seconds"),
        "dbt": {"project_name": root_pkg, "dbt_version": m["metadata"].get("dbt_version")},
        "dependency_fingerprint": C.dependency_fingerprint(str(project_dir)),
        "stats": {
            "models_in_graph": sum(n["models"] for n in out_nodes),
            "tests_in_graph": sum(n["tests"] for n in out_nodes),
            "tasks": len(out_nodes),
            "edges": len(reduced),
            "edges_before_reduction": len(node_edges),
            "merged_cycle_nodes": [n["id"] for n in out_nodes if n["merged_cycle"]],
            "disabled_models": sum(1 for v in m["disabled"].values() for x in v if x["resource_type"] == "model"),
        },
        "nodes": sorted(out_nodes, key=lambda n: (n["group"], n["task"])),
        "edges": sorted([list(e) for e in reduced]),
        "sources": sources,
        "extractors": extractors,
    }


def main() -> None:
    ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("--project-dir", default=C.DBT_PROJECT_DIR)
    ap.add_argument("--manifest", help="existing manifest.json (default: <project>/target/manifest.json)")
    ap.add_argument("--parse", action="store_true", help="run `dbt deps` + `dbt parse` first")
    ap.add_argument("--profiles-dir", default=None, help="default: <project>/DEV_PROFILES")
    ap.add_argument("--config-dir", default=C.CONFIG_DIR)
    ap.add_argument("--out", default=C.GRAPH_PATH)
    a = ap.parse_args()

    project = Path(a.project_dir).resolve()
    manifest = Path(a.manifest) if a.manifest else project / "target" / "manifest.json"
    if a.parse:
        manifest = run_parse(project, a.profiles_dir or str(project / "DEV_PROFILES"))
    if not manifest.is_file():
        sys.exit(f"manifest not found: {manifest} (use --parse or --manifest)")

    graph = build(manifest, project, Path(a.config_dir))
    Path(a.out).parent.mkdir(parents=True, exist_ok=True)
    Path(a.out).write_text(json.dumps(graph, indent=1, sort_keys=False) + "\n", encoding="utf-8")
    s = graph["stats"]
    print(f"wrote {a.out}: {s['tasks']} tasks, {s['edges']} edges "
          f"({s['edges_before_reduction']} before reduction), {s['models_in_graph']} models, {s['tests_in_graph']} tests")
    if s["merged_cycle_nodes"]:
        print("merged to break group-level cycles:", ", ".join(s["merged_cycle_nodes"]))


if __name__ == "__main__":
    main()
